ξεγελούν προς τα έξω.

# αναγνωριστικό αρχείου → (όνομα εικονιδίου, ζουμ, κατακόρυφη μετατόπιση)
CATALOGUE = {

    'a3qs0b': ('angel', 2.05, -0.02),

}


def build(destination: str) -> list[tuple[str, int]]:
    os.makedirs(destination, exist_ok=True)
    built = []
    for entry in sorted(os.listdir(SOURCE)):
        if not entry.lower().endswith('.png') or entry.startswith('_'):
            continue
        key = next((k for k in CATALOGUE if k in entry), None)
        if key is None:
            print(f'ΑΓΝΩΣΤΟ ΑΡΧΕΙΟ: {entry}')
            continue
        name, zoom, shift = CATALOGUE[key]
        target = os.path.join(destination, f'{name}.webp')
        prepare(os.path.join(SOURCE, entry), target, zoom, shift)
        built.append((name, os.path.getsize(target) // 1024))
    return built


if __name__ == '__main__':
    for name, kilobytes in build(sys.argv[1]):
        print(f'{name:16} {kilobytes:3} KB')
