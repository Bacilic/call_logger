// ΠΡΟΣΩΡΙΝΟ: παράγει το assets/servers.png. Διαγράφεται μετά την παραγωγή.
//   flutter test test/_icon_gen_test.dart

import 'dart:io';
import 'dart:typed_data';
import 'dart:ui' as ui;

import 'package:flutter_test/flutter_test.dart';

const _dark = ui.Color(0xFF4C6472);
const _light = ui.Color(0xFFAFC4D0);
const _green = ui.Color(0xFF3FA34D);
const _white = ui.Color(0xFFFFFFFF);

void main() {
  testWidgets('παράγει το εικονίδιο διακομιστών', (tester) async {
    await tester.runAsync(() async {
      final recorder = ui.PictureRecorder();
      final canvas = ui.Canvas(recorder);

      const unitW = 292.0;
      const unitH = 76.0;
      const gap = 16.0;
      const left = 64.0;
      const top = 72.0;

      ui.RRect unit(int index) => ui.RRect.fromRectAndRadius(
        ui.Rect.fromLTWH(left, top + index * (unitH + gap), unitW, unitH),
        const ui.Radius.circular(16),
      );

      // Δύο πίσω μονάδες σε ανοιχτό τόνο, η μπροστινή σκούρα — ίδια ιεραρχία
      // με τα υπόλοιπα εικονίδια του hub.
      final lightPaint = ui.Paint()..color = _light;
      canvas.drawRRect(unit(0), lightPaint);
      canvas.drawRRect(unit(1), lightPaint);
      canvas.drawRRect(unit(2), ui.Paint()..color = _dark);

      // Λεπτομέρειες μόνο στη σκούρα: ενδεικτική λυχνία και σχισμές αερισμού.
      final whitePaint = ui.Paint()..color = _white;
      final frontCenterY = top + 2 * (unitH + gap) + unitH / 2;
      canvas.drawCircle(ui.Offset(left + 38, frontCenterY), 10, whitePaint);
      for (final dy in [-12.0, 12.0]) {
        canvas.drawRRect(
          ui.RRect.fromRectAndRadius(
            ui.Rect.fromLTWH(left + 72, frontCenterY + dy - 4.25, 118, 8.5),
            const ui.Radius.circular(4.25),
          ),
          whitePaint,
        );
      }

      // Το σήμα λειτουργίας ακουμπά τη ΓΩΝΙΑ της μπροστινής μονάδας, δεν την
      // κόβει στη μέση: πρώτα ανοίγει κενό (clear), μετά μπαίνει ο δίσκος.
      const badgeCenter = ui.Offset(386, 378);
      canvas.drawCircle(
        badgeCenter,
        72,
        ui.Paint()..blendMode = ui.BlendMode.clear,
      );
      canvas.drawCircle(badgeCenter, 62, ui.Paint()..color = _green);

      // Σύμβολο power: τόξο ανοιχτό στην κορυφή + κάθετη γραμμή.
      final arcPaint = ui.Paint()
        ..color = _white
        ..style = ui.PaintingStyle.stroke
        ..strokeWidth = 12
        ..strokeCap = ui.StrokeCap.round;
      canvas.drawArc(
        ui.Rect.fromCircle(center: badgeCenter, radius: 28),
        -1.15,
        4.9,
        false,
        arcPaint,
      );
      canvas.drawLine(
        badgeCenter.translate(0, -36),
        badgeCenter.translate(0, -6),
        arcPaint,
      );

      final image = await recorder.endRecording().toImage(512, 512);
      final data = await image.toByteData(format: ui.ImageByteFormat.png);
      final bytes = Uint8List.view(data!.buffer);
      File('assets/servers.png').writeAsBytesSync(bytes);
      image.dispose();
    });
  });
}
